  <!-- scripts: lens + card tabs + scroll reveal -->
  <script>
    (function(){
      // lens tracking
      const lens = document.getElementById('magnifyingLens');
      let mouseX = -9999, mouseY = -9999;
      
      document.addEventListener('mousemove', (e) => {
        mouseX = e.clientX;
        mouseY = e.clientY;
        lens.style.left = mouseX + 'px';
        lens.style.top = mouseY + 'px';
      });

      // optional: hide lens on touch devices
      if('ontouchstart' in window) {
        lens.style.display = 'none';
      }

      // card stack tabs
      const cards = document.querySelectorAll('.stack-card');
      const indicators = document.querySelectorAll('.tab-indicator');
      
      function setActive(index) {
        cards.forEach((c, i) => {
          c.classList.toggle('active', i === index);
        });
        indicators.forEach((dot, i) => {
          dot.classList.toggle('active', i === index);
        });
        // update counter
        const counter = document.querySelector('.tab-indicator + span');
        if (counter) {
          const num = String(index + 1).padStart(2, '0');
          counter.textContent = num + ' / 03';
        }
      }

      indicators.forEach((dot) => {
        dot.addEventListener('click', () => {
          const target = parseInt(dot.dataset.target);
          setActive(target);
        });
      });

      // auto rotate cards every 5s
      let currentIdx = 0;
      setInterval(() => {
        currentIdx = (currentIdx + 1) % 3;
        setActive(currentIdx);
      }, 5000);

      // scroll reveal using Intersection Observer
      const revealEl = document.getElementById('revealSection');
      const observer = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            entry.target.classList.add('visible');
          }
        });
      }, { threshold: 0.2 });
      if (revealEl) observer.observe(revealEl);

      // additionally, if lens should auto-animate? we keep it cursor driven.
      // add subtle gradient movement on hero? optional.

      // fix for small screens: adjust lens size
      function handleResize() {
        if (window.innerWidth < 640) {
          lens.style.width = '120px';
          lens.style.height = '120px';
        } else {
          lens.style.width = '200px';
          lens.style.height = '200px';
        }
      }
      window.addEventListener('resize', handleResize);
      handleResize();
    })();
  </script>
